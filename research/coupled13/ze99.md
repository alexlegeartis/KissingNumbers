import numpy as np, re
S3 = np.sqrt(3.0)
def pc(tok):
    tok = tok.strip()
    if 'sqrt' in tok:
        m = re.match(r'^(-?\d*)\*?sqrt\(3\)$', tok)
        c = m.group(1)
        if c in ('','+'): co=1.0
        elif c=='-': co=-1.0
        else: co=float(c)
        return co * S3
    return float(tok)
